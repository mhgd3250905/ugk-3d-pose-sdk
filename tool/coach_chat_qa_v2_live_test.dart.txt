class _ObservedClient implements coach.LlmClient {
  _ObservedClient(this.inner, this.calls);
  final coach.OpenAiCompatibleClient inner;
  final List<Map<String, Object?>> calls;
  @override String get model => inner.model;
  @override coach.Pricing? get pricing => inner.pricing;
  @override Future<coach.LlmResponse> complete(coach.LlmRequest request, {coach.CancelToken? cancel}) async {
    final watch = Stopwatch()..start();
    try {
      final response = await inner.complete(request, cancel: cancel);
      calls.add({'ms': watch.elapsedMilliseconds, 'tools': response.toolCalls.map((c) => c.name).toList(),
        'usage': {'input': response.usage.promptTokens, 'output': response.usage.completionTokens}});
      return response;
    } on Object catch (error) {
      calls.add({'ms': watch.elapsedMilliseconds, 'errorType': error.runtimeType.toString()});
      rethrow;
    }
  }
}
