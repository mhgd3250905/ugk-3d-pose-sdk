    jsonDecode(File(path).readAsStringSync()) as Map<String, dynamic>;

void main() {
  test('one fresh SF conversation through CoachChatController', () async {
    final env = Platform.environment;
    final output = env['CHAT_EXTRACT_LIVE_OUTPUT']!;
    final sdk = env['CHAT_EXTRACT_LIVE_SDK']!;
    final source = env['CHAT_EXTRACT_LIVE_SOURCE']!;
    final dry = env['CHAT_EXTRACT_LIVE_MODE'] == 'dry';
    final thinking = thinking_mode.parseEvaluationThinkingMode(
      env['CHAT_EXTRACT_LIVE_THINKING']);
    final profile = coach.CoachProfile.fromJson(
      readJson('$source/assets/coach/smartfit.json'));
    final deliveryPath = '$sdk/packages/ugk_trainer/assets/pose_hub/app-test-pack-r1';
    final delivery = readJson('$deliveryPath/catalog.json');
    final metadata = coach.PoseHubAppTestCatalog.fromJson(delivery);
    final sdkCatalog = coach.PoseHubAppTestCatalog.fromJson(delivery,
      semanticCatalog: readJson('$deliveryPath/${metadata.semanticCatalogFile.path}'));
    final content = sf_content.AppTestContent.build(
      catalog: sf_catalog.AppTestCatalog.fromSdk(sdkCatalog),

    final catalog = content.coachCatalog;
    // This exact instance supplies both the profile callback and measurements.
    final builder = trainer.SessionBuilder(prepSeconds: 10,
      rhythms: sf_session.trainerRhythms(content.library), known: content.library.contains);
    final actualSfPlayer = sf_session.SessionBuilder(content.library);
    final allowed = (readJson('$sdk/test/fixtures/action-scope/smartfit-219.json')
      ['allowedActionIds'] as List).cast<String>();
    final manifest = coach.CoachActionScopeManifest.fromCatalogBytes(appId: 'smartfit',
      catalogBytes: File('$deliveryPath/catalog.json').readAsBytesSync(),

    var nativeTimelineCompared = false;
    Json measure(coach.TrainingPlan plan) {
      final native = actualSfPlayer.phases(sf_plan.TrainingPlan.fromJson(plan.toJson()));
      final phases = builder.phases(plan);
      if (native.length != phases.length) throw StateError('SF phase count differs');
      for (var index = 0; index < phases.length; index++) {
        final a = native[index], b = phases[index];
        if (a.kind != b.kind || a.seconds != b.seconds || a.exerciseId != b.exerciseId ||
            a.blockIndex != b.blockIndex || a.set != b.set || a.round != b.round ||
            a.side != b.side || a.selfPaced != b.selfPaced || a.segmentId != b.segmentId) {
          throw StateError('SF phase $index differs');
        }
      }
      final timing = coach.evaluatePlanExecutionTiming(plan, builder.executionTiming);
      final nativeSeconds = native.fold<double>(0, (total, phase) => total + phase.seconds);
      if ((nativeSeconds - timing.totalSeconds).abs() > 1e-9) {
        throw StateError('SF player and callback totals differ');
      }
      nativeTimelineCompared = true;
      final card = const coach.PlanDuration().estimateSeconds(plan, catalog);
      final target = plan.targetMinutes * 60;
      return {
        'targetSeconds': target, 'cardSeconds': card,
        'executionSeconds': timing.totalSeconds, 'nativeSeconds': nativeSeconds,
        'cardDeviationPercent': (card - target) / target * 100,
        'executionDeviationPercent': (timing.totalSeconds - target) / target * 100,
        'isEstimate': timing.containsManualCounting,
        'manualCountingLabel': timing.containsManualCounting ? '估算' : null,

    final failures = <String>[];
    Object? fatal;
    Json? sample;

      final initialized = await controller.start();
      if (!initialized.ok) throw StateError('Chat startup failed: ${initialized.error.runtimeType}');
      if (dry) {
        expect(HttpOverrides.current, isNull,

        final drill = catalog.drills.firstWhere((d) => controller.agent.actionScope!.contains(d.id));

          final problems = <String>[];
          Json? timing;

          if (index == 1) {

          } else if (plan == null || plan.items.isEmpty) {

              for (final key in ['cardSeconds', 'executionSeconds']) {
                final seconds = timing[key]! as num;

            } on Object catch (error) {
              timingError = error.runtimeType.toString();
              problems.add('native_timeline_measurement_failed');
            }
          }
          final calls = wire!.calls.sublist(callStart);
          final row = <String, Object?>{
            'turn': index + 1, 'input': input, 'status': result?.status.name,
            'text': result?.text, 'errorType': result?.error?.runtimeType.toString(),
            'elapsedMs': watch.elapsedMilliseconds, 'plan': plan?.toJson(),
            'actionIds': actions, 'timing': timing, 'timingError': timingError,
            'httpCalls': calls.length,
            'httpElapsedMs': calls.fold<num>(0, (sum, call) => sum + (call['elapsedMs'] as num? ?? 0)),
            'maxRequestBytes': calls.isEmpty ? null
              : calls.map((call) => call['requestBytes']! as int).reduce((a, b) => a > b ? a : b),
            'usage': _usage(calls),
            'issueCodes': result == null ? <String>[]
              : {for (final record in result.records) ...record.issueCodes}.toList(),
            'steps': result?.steps,
            'failures': problems, 'passed': problems.isEmpty,
          };

Json _usage(List<Json> calls) {
  final result = <String, Object?>{};
  for (final key in ['prompt_tokens', 'completion_tokens', 'total_tokens', 'cached_tokens']) {
    final values = [for (final call in calls)
      if ((call['usage'] as Map?)?[key] case final num value) value];
    result[key] = {
      'total': values.isEmpty ? null : values.fold<num>(0, (a, b) => a + b),
      'knownCalls': values.length, 'unknownCalls': calls.length - values.length,
    };
  }
  return result;
}
